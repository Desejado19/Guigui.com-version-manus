CREATE TABLE IF NOT EXISTS reviews (
  id TEXT PRIMARY KEY,
  business_id TEXT NOT NULL,
  author_user_id TEXT REFERENCES users(id) ON DELETE SET NULL,
  author_name TEXT NOT NULL,
  author_phone_hash TEXT NOT NULL,
  author_phone_masked TEXT NOT NULL,
  qualitative_score TEXT NOT NULL CHECK (qualitative_score IN ('nao_gostei', 'gostei', 'gostei_muito', 'muito_bom')),
  score_weight SMALLINT NOT NULL CHECK (score_weight BETWEEN 1 AND 4),
  comment TEXT CHECK (char_length(comment) <= 200),
  status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('approved', 'pending', 'flagged')),
  moderation_reason TEXT,
  owner_reply TEXT,
  verified_sms BOOLEAN NOT NULL DEFAULT FALSE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS reviews_business_idx ON reviews(business_id, created_at DESC);
CREATE INDEX IF NOT EXISTS reviews_status_idx ON reviews(status, created_at DESC);

#!/usr/bin/env tsx
/**
 * Guigui.com â€” MigraÃ§Ã£o do esquema PostgreSQL
 *
 * Executa, por ordem alfabÃ©tica, todos os ficheiros SQL de `db/migrations/`
 * contra a base de dados configurada em DATABASE_URL. Todas as migraÃ§Ãµes sÃ£o
 * idempotentes (IF NOT EXISTS / IF EXISTS), pelo que repetir a execuÃ§Ã£o Ã© seguro.
 *
 * Uso:
 *   npm run db:migrate
 *
 * Requisitos:
 *   - VariÃ¡vel de ambiente DATABASE_URL definida (.env)
 *   - Servidor PostgreSQL acessÃ­vel
 */
import 'dotenv/config';
import { Pool } from 'pg';
import fs from 'fs';
import path from 'path';

const DATABASE_URL = process.env.DATABASE_URL;
if (!DATABASE_URL) {
  console.error('ERRO: DATABASE_URL nÃ£o estÃ¡ definida no ambiente (.env).');
  console.error('Defina DATABASE_URL=postgresql://user:password@localhost:5432/guigui e volte a executar.');
  process.exit(1);
}

const pool = new Pool({ connectionString: DATABASE_URL });

async function main() {
  let client;
  try {
    client = await pool.connect();
    console.log('[db:migrate] Conectado ao PostgreSQL.');

    const migrationsDir = path.resolve(process.cwd(), 'db', 'migrations');
    const files = fs.readdirSync(migrationsDir).filter((file) => file.endsWith('.sql')).sort();
    console.log(`[db:migrate] ${files.length} migraÃ§Ã£o(Ãµes) encontrada(s).`);

    for (const file of files) {
      const schemaSql = fs.readFileSync(path.join(migrationsDir, file), 'utf8');
      // Executa o ficheiro inteiro (client.query aceita mÃºltiplas statements separadas por ;)
      await client.query(schemaSql);
      console.log('[db:migrate] Aplicada:', file);
    }

    // VerificaÃ§Ã£o rÃ¡pida: lista tabelas criadas
    const tables = await client.query(
      `SELECT table_name FROM information_schema.tables
       WHERE table_schema='public' AND table_type='BASE TABLE'
       ORDER BY table_name`
    );
    console.log(`[db:migrate] Tabelas na base de dados (${tables.rows.length}):`);
    for (const row of tables.rows) {
      console.log(`  - ${row.table_name}`);
    }
  } catch (error) {
    console.error('[db:migrate] Falha na migraÃ§Ã£o:', error instanceof Error ? error.message : error);
    process.exit(1);
  } finally {
    if (client) client.release();
    await pool.end();
  }
}

main();
